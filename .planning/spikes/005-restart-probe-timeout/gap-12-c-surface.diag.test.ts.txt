// GAP-12-C DIAGNOSIS-ONLY probe (renderer half) — does a live→failed-restart error
// notice SURFACE or get SWALLOWED?
//
// Traces applyStatusEvent for the exact event create() broadcasts on a bad-cwd restart:
//   { status: 'error', notice: 'Working directory not found: <bad>' }
// against a row that was LIVE ('running') when the restart was attempted.
//
// Throwaway artifact for the debug session; the fix plan removes/relocates it.

import { describe, it, expect } from 'vitest';
import { applyStatusEvent, type StatusRow } from '../apply-status-event';
import type { LogicalId } from '../../shared/types';

function runningRow(): StatusRow {
  return {
    logicalId: 'live-1' as LogicalId,
    ptyPid: 4242,
    name: 'API',
    icon: { type: 'emoji', value: '🖥️' },
    cwd: '/Users/dev/proj',
    shell: '/bin/zsh',
    startupCommand: 'echo hi',
    status: 'running',
    order: 0,
    lastActive: 1,
    configured: true,
  };
}

describe('GAP-12-C — does the bad-cwd restart error SURFACE on a LIVE row? (confirm/deny)', () => {
  it('an error event WITH a notice is treated as INFORMATIONAL — status stays running, NO IdleCard', () => {
    const row = runningRow();
    // This is exactly what create() sends on a pid -1 bad-cwd spawn (status:'error'
    // + the 'Working directory not found' notice on the same onPtyStatus event).
    const next = applyStatusEvent(row, {
      status: 'error',
      notice: 'Working directory not found: /no/such/dir',
    });

    // The ITEM-4 guard (notice = informational, never a lifecycle transition) keeps
    // the row's CURRENT status. So the row stays 'running' — NOT 'error'/'not_started'.
    expect(next.status).toBe('running'); // status NOT flipped to error
    // The errorMessage IS captured on the row...
    expect(next.errorMessage).toBe('Working directory not found: /no/such/dir');
    // ...but because status stays 'running', SessionManager keeps the SessionView mounted
    // (startedSessions filter = status !== not_started && !== error) and does NOT render
    // the IdleCard error branch (activeIsCard = not_started || error). So the captured
    // errorMessage has NO visible surface unless the user reopens the edit modal.
    const activeIsCard = next.status === 'not_started' || next.status === 'error';
    expect(activeIsCard).toBe(false); // no IdleCard → error message is not shown
    // ptyPid is unchanged (handleRestart did NOT flip on pid -1; the notice branch does
    // not touch ptyPid either) — the row still claims a now-dead pid.
    expect(next.ptyPid).toBe(4242);
  });

  it('CONTROST: a bare error transition WITHOUT a notice DOES flip an identity row to the Inactive List', () => {
    const row = runningRow();
    // A lifecycle 'error' with NO notice (the abnormal fork-then-die path sends the
    // notice separately) flips an identity row to not_started → Inactive List.
    const next = applyStatusEvent(row, { status: 'error' });
    expect(next.status).toBe('not_started'); // identity self-exit → Inactive List
    expect(next.ptyPid).toBeUndefined();
  });
});
